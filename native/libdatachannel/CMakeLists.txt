cmake_minimum_required(VERSION 3.21)
project(BeutlDataChannel LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_DEPS_LIBS OFF CACHE BOOL "" FORCE)
set(USE_MBEDTLS ON CACHE BOOL "" FORCE)
set(NO_WEBSOCKET ON CACHE BOOL "" FORCE)
set(NO_EXAMPLES ON CACHE BOOL "" FORCE)
set(NO_TESTS ON CACHE BOOL "" FORCE)
# libdatachannel's find module omits X509's dependency on Crypto, leaving
# certificate-writing symbols unresolved with a single-pass static linker.
file(TO_CMAKE_PATH "${DATACHANNEL_SOURCE}" DATACHANNEL_SOURCE)
list(APPEND CMAKE_MODULE_PATH "${DATACHANNEL_SOURCE}/cmake/Modules")
find_package(MbedTLS 3 REQUIRED MODULE)
set_property(TARGET MbedTLS::MbedX509 APPEND PROPERTY INTERFACE_LINK_LIBRARIES MbedTLS::MbedCrypto)
add_subdirectory(${DATACHANNEL_SOURCE} datachannel EXCLUDE_FROM_ALL)
add_library(BeutlDataChannel SHARED preview.cpp)
target_link_libraries(BeutlDataChannel PRIVATE datachannel-static)
set_target_properties(BeutlDataChannel PROPERTIES MACOSX_RPATH ON)
if(APPLE)
  target_link_options(BeutlDataChannel PRIVATE "-Wl,-dead_strip")
elseif(UNIX)
  target_link_options(BeutlDataChannel PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(preview-packetizer-tests packetizer_test.cpp)
  target_link_libraries(preview-packetizer-tests PRIVATE datachannel-static)
  add_test(NAME preview-packetizer COMMAND preview-packetizer-tests)
endif()
install(TARGETS BeutlDataChannel RUNTIME DESTINATION . LIBRARY DESTINATION .)
